\begin{frame}{Opportunity: Conditional Jacobian}
\begin{columns}[T]
    \begin{column}{0.55\textwidth}
        \textbf{When is Jacobian Needed?}
        \begin{itemize}
            \item Multi-Branch Fit (MBF) smoother requires accumulated Jacobian
            \item Used for backward smoothing pass
            \item \textbf{Scenario:} When MBF disabled $\rightarrow$ Jacobian computed but \textcolor{acadred}{never used}
        \end{itemize}
    
        \vspace{1em}
        \textbf{Current Behavior} (\texttt{parameter\_transporter})
        \begin{enumerate}
            \item Compute full 6$\times$6 Jacobian at each surface
            \item Multiply: $J_{acc} = J_{new} \times J_{acc}$
            \item Store accumulated Jacobian to global memory
        \end{enumerate}
    \end{column}

    \begin{column}{0.45\textwidth}
        \centering
        \begin{tikzpicture}[scale=0.8]
            % Track line
            \draw[thick,->, >=stealth, color=gray!80] (0,0) -- (6,0) node[right, black] {Track};

\begin{frame}{Backup: FAQ}
    \textbf{Q1: Why does Jacobian computation still happen?}
    \begin{itemize}
        \item \texttt{get\_full\_jacobian()} is needed for covariance transport. We only skip the \textit{aggregation}.
    \end{itemize}

    \vspace{0.5em}
    \textbf{Q2: Why not runtime dispatch?}
    \begin{itemize}
        \item \texttt{if constexpr} enables dead code elimination and specific register allocation.
    \end{itemize}
    
    \vspace{0.5em}
    \textbf{Q3: Why different results on sm\_70 vs sm\_75?}
    \begin{itemize}
        \item Compiler optimizations, instruction sets (tensor cores), and register allocation heuristics vary by architecture.
    \end{itemize}
\end{frame}

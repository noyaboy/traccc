\section{Problem Context}
%==============================================================================%

\begin{frame}{Register Pressure in GPU Track Finding}
\begin{columns}
\column{0.5\textwidth}
\textbf{The Problem}
\begin{itemize}
    \item \texttt{propagate\_to\_next\_surface} kernel
    \item Only \textbf{10-25\% GPU occupancy}
    \item Root cause: High register usage (128-203 registers/thread)
    \item V100 needs $\leq$32 registers for 100\% occupancy
\end{itemize}

\vspace{0.5em}
\textbf{Impact}
\begin{itemize}
    \item SM underutilization
    \item Poor latency hiding
    \item Reduced throughput
\end{itemize}

\column{0.5\textwidth}
\centering
\textbf{Register Budget Analysis}
\begin{table}
    \scriptsize
    \begin{tabular}{lc}
        \toprule
        \textbf{Component} & \textbf{Registers} \\
        \midrule
        Bound parameters (6 floats) & 6 \\
        Covariance (6$\times$6 symmetric) & 21 \\
        \textbf{Jacobian (6$\times$6)} & \textbf{$\sim$36} \\
        RK4 derivatives & 12 \\
        B-field vectors & 9 \\
        Navigation state & 8-12 \\
        Actor states & 20-30 \\
        \midrule
        \textbf{Total} & \textbf{120-150} \\
        \bottomrule
    \end{tabular}
\end{table}

\begin{alertblock}{Key Insight}
    Jacobian is only needed for MBF smoother!
\end{alertblock}
\end{columns}
\end{frame}

\begin{frame}{Opportunity: Conditional Jacobian}
\begin{columns}
\column{0.55\textwidth}
\textbf{When is Jacobian Needed?}
\begin{itemize}
    \item Multi-Branch Fit (MBF) smoother requires accumulated Jacobian
    \item Used for backward smoothing pass
    \item When MBF disabled: Jacobian computed but \textbf{never used}
\end{itemize}

\vspace{0.5em}
\textbf{Current Behavior (parameter\_transporter)}
\begin{enumerate}
    \item Compute full 6$\times$6 Jacobian at each surface
    \item Multiply: $J_{acc} = J_{new} \times J_{acc}$ (6$\times$6 $\times$ 6$\times$6)
    \item Store accumulated Jacobian to global memory
    \item \textcolor{red}{Wasted work when MBF=false!}
\end{enumerate}

\begin{frame}[fragile]{Key Code: Actor State Comparison}
\begin{columns}
\column{0.5\textwidth}
\textbf{parameter\_transporter (detray)}
\begin{lstlisting}[language=C++]
// State contains Jacobian pointer
struct state {
    bound_matrix_t* _full_jacobian_ptr
        = nullptr;
};

// In operator():
if (actor_state._full_jacobian_ptr
    != nullptr) {
    // 6x6 x 6x6 matrix multiplication
    const auto aggregate =
        full_jacobian *
        (*(actor_state._full_jacobian_ptr));
    // Write back to global memory
    (*(actor_state._full_jacobian_ptr)) =
        aggregate;
}
\end{lstlisting}

\column{0.5\textwidth}
\textbf{bound\_updater (traccc)}
\begin{lstlisting}[language=C++]
// Empty state - no Jacobian pointer!
struct state {};

// In operator():
// Compute full_jacobian (same as before)
const auto full_jacobian =
    get_full_jacobian(propagation);

// Transport covariance (same as before)
transport_covariance_to_bound_impl(
    old_cov, full_jacobian, new_cov);

// NOTE: No Jacobian aggregation!
// This is the key difference.
\end{lstlisting}
\end{columns}

\vspace{0.3em}
\begin{alertblock}{Key Difference}
    \texttt{bound\_updater} computes Jacobian for covariance transport but \textbf{skips the 6$\times$6 matrix multiplication and global memory write}.
\end{alertblock}
\end{frame}

\begin{frame}[fragile]{Kernel Dispatch Logic}
\begin{columns}
\column{0.55\textwidth}
\textbf{propagate\_to\_next\_surface.ipp}
\begin{lstlisting}[language=C++]
// Compile-time check for actor chain type
if constexpr (
    details::has_jacobian_transport_v<
        propagator_t>) {
    // 7-actor chain with param_transporter
    // Initialize Jacobian for MBF smoother
    payload.tmp_jacobian_ptr[param_id] =
        matrix::identity<bound_matrix_t>();
    s1._full_jacobian_ptr =
        &payload.tmp_jacobian_ptr[param_id];

    propagator.propagate(propagation,
        detray::tie(s0,s1,s2,s3,s4,s5,s6));
} else {
    // 7-actor chain with bound_updater
    // No Jacobian initialization needed

    propagator.propagate(propagation,
        detray::tie(s0,s1,s2,s3,s4,s5,s6));
}
\end{lstlisting}

\column{0.45\textwidth}
\textbf{Type Trait Detection}
\begin{lstlisting}[language=C++]
/// Check if propagator uses Jacobian
template <typename propagator_t>
struct has_jacobian_transport
    : contains_type<
        parameter_transporter<algebra_t>,
        typename propagator_t::
            actor_chain_type::actor_tuple
    > {};

template <typename propagator_t>
inline constexpr bool
    has_jacobian_transport_v =
    has_jacobian_transport<
        propagator_t>::value;
\end{lstlisting}

\begin{frame}{Methodology: Isolating the True Improvement}
\begin{columns}
\column{0.5\textwidth}
\textbf{Initial Result: +11.67\%}
\begin{itemize}
    \item First benchmark showed 38.75 $\rightarrow$ 43.27 ev/s
    \item But \texttt{build\_tracks} improved by \textbf{-86.3\%}!
    \item Something else was happening...
\end{itemize}

\textbf{Discovery: Configuration Change}
\begin{itemize}
    \item Baseline: \texttt{run\_mbf\_smoother = true}
    \item Optimization: \texttt{run\_mbf\_smoother = false}
    \item This was a \textbf{default value change}, not just the actor swap
\end{itemize}

\textbf{Confounded Effects}
\begin{table}
    \scriptsize
    \begin{tabular}{lc}

\begin{frame}{Backup: Anticipated Questions (1/2)}
\textbf{Q1: Why does Jacobian computation still happen?}
\begin{itemize}
    \item \texttt{get\_full\_jacobian()} is needed for covariance transport
    \item We skip the \textit{aggregation}, not the computation
    \item Future work: skip computation entirely when not needed
\end{itemize}

\vspace{0.5em}
\textbf{Q2: Why not runtime dispatch instead of compile-time?}
\begin{itemize}
    \item \texttt{if constexpr} allows dead code elimination
    \item Runtime \texttt{if} would still allocate registers for Jacobian pointer
    \item Compile-time dispatch enables different kernel specializations
\end{itemize}

\vspace{0.5em}
\textbf{Q3: What is MBF smoother and when is it needed?}
\begin{itemize}
    \item Multi-Branch Fit smoother: backward pass to improve track parameters
    \item Needed for high-precision physics measurements
    \item Can be disabled for real-time trigger applications
\end{itemize}
\end{frame}

\begin{frame}{Backup: Anticipated Questions (2/2)}
\textbf{Q4: Why different results on sm\_70 vs sm\_75?}
\begin{itemize}
    \item CUDA compiler makes architecture-specific optimizations
    \item Different instruction sets (sm\_75 has tensor cores)
    \item Register allocation heuristics vary by architecture
    \item Always profile on production hardware
\end{itemize}

\vspace{0.5em}
\textbf{Q5: What about other register-heavy kernels?}
\begin{itemize}
    \item \texttt{fit\_forward}: 168 registers
    \item \texttt{fit\_backward}: 168 registers
    \item Similar conditional optimization could apply
    \item Future work opportunity
\end{itemize}

\vspace{0.5em}
\textbf{Q6: How does this compare to other optimizations?}
\begin{itemize}
    \item Multi-event batching: +93\% throughput
    \item Conditional Jacobian: +18.3\% throughput
    \item Complementary optimizations, can be combined
\end{itemize}
\end{frame}

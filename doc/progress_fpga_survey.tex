%--- FP32 vs FP64 ---%
\begin{frame}{1. FP32 vs FP64 Precision Validation}
\textbf{What I did:}
\begin{itemize}
    \item Built TRACCC with \texttt{-DTRACCC\_CUSTOM\_SCALARTYPE=float} and \texttt{=double}
    \item Ran Kalman fitter tests on 10,000 muon tracks (1 GeV, 10 GeV configurations)
    \item Compared pull distributions ($\mu$, $\sigma$) and $\chi^2$/NDF
\end{itemize}

\vspace{0.5em}
\textbf{What it means:}
\begin{itemize}
    \item \textbf{Result:} FP32 and FP64 produce \textbf{identical} physics results
    \item Pull $\sigma$ differs by $<0.003$ (within statistical noise)
    \item $\chi^2$/NDF = $1.0042 \pm 0.0020$ for both precisions
    \item \textbf{Implication:} DSP58 native FP32 is safe for FPGA offloading --- no precision loss
\end{itemize}
\end{frame}

%--- NCU Profiling ---%
\begin{frame}{2. NCU Profiling Analysis}
\textbf{What I did:}
\begin{itemize}
    \item Profiled \texttt{propagate\_to\_next\_surface} kernel with NVIDIA Nsight Compute
    \item Analyzed register pressure, cache hit rates, warp stall reasons
\end{itemize}

\vspace{0.5em}
\textbf{What it means:}
\begin{itemize}
    \item \textbf{93\% warp stall cycles} --- kernel is latency-bound, not compute-bound
    \item \textbf{96--128 registers/thread} --- high register pressure limits occupancy
    \item \textbf{46--54\% L1 cache hit} --- unpredictable memory access patterns
    \item \textbf{Implication:} GPU's SIMT model is fundamentally inefficient for this workload. FPGA's pipelined execution can eliminate stalls.
\end{itemize}
\end{frame}

%--- FPGA Suitability ---%
\begin{frame}{4. FPGA Suitability Assessment}
\textbf{What I did:}
\begin{itemize}
    \item Analyzed each kernel's computational pattern (MAC chains, memory access, branching)
    \item Mapped operations to DSP58 capabilities
\end{itemize}

\vspace{0.5em}
\textbf{What it means:}
\begin{columns}
\column{0.5\textwidth}
\textbf{Good for FPGA:}
\begin{itemize}
    \item RK4 propagation (MAC chains)
    \item B-field polynomial (Horner's method)
    \item Matrix-vector ops (systolic arrays)
    \item CCL clustering (well-known FPGA pattern)
\end{itemize}

\column{0.5\textwidth}
\textbf{Keep on GPU:}
\begin{itemize}
    \item Track deduplication (irregular branching)
    \item build\_tracks (pointer chasing)
    \item 6$\times$6 matrix inversion (complex control)
\end{itemize}
\end{columns}

\vspace{0.5em}
\textbf{Implication:} Clear partitioning strategy --- offload compute-regular kernels, keep irregular ones on GPU.
\end{frame}

\begin{frame}{6. PCIe Latency Analysis}
\textbf{What I did:}
\begin{itemize}
    \item Measured GPU$\leftrightarrow$Host transfer latency as proxy for GPU$\leftrightarrow$FPGA
    \item Tested different data sizes (params only vs full state vs + Jacobians)
    \item \textbf{Limitation:} Proxy only --- actual GPU$\leftrightarrow$FPGA differs due to XRT overhead and PCIe topology
\end{itemize}

\vspace{0.5em}
\textbf{What it means:}
\begin{table}
\small
\begin{tabular}{lcl}
    \toprule
    \textbf{Transfer Scenario} & \textbf{Overhead} & \textbf{Assessment} \\
    \midrule
    Parameters only (24 B/track) & 2.7\% & \cmark Acceptable \\

    \bottomrule

\textbf{Implication:} Transfer strategy defined (params-only), but actual overhead needs validation on real V80 hardware.
\end{frame}

\section{Next Steps}
%==============================================================================%

\begin{frame}{Next Step: Validate Per-Step Sync Barrier}
\begin{alertblock}{Critical Blocker}
CKF algorithm requires GPU$\leftrightarrow$FPGA synchronization at \textbf{each of 15 surfaces}. If sync overhead $>$ 200$\mu$s/step, FPGA offloading is \textbf{not viable}.
\end{alertblock}

\vspace{0.5em}
\textbf{What needs to be done:}
\begin{enumerate}
    \item Instrument CKF loop to measure actual per-step synchronization time
    \item Test with GPU$\leftrightarrow$Host sync as proxy (no FPGA hardware yet)
    \item Calculate total overhead: $15 \times \text{sync\_time}$ vs 23ms baseline
\end{enumerate}

\vspace{0.5em}
\textbf{Decision criteria:}
\begin{itemize}
    \item If overhead $<$ 5\% ($<$1.15ms total): \textcolor{acadgreen}{Proceed with FPGA development}
    \item If overhead 5--15\%: Consider async/overlap strategies
    \item If overhead $>$ 15\%: \textcolor{acadred}{FPGA offloading not viable for CKF}
\end{itemize}

\vspace{0.5em}
\textbf{This is the go/no-go decision point before any FPGA development.}
\end{frame}

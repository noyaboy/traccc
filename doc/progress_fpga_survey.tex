%--- Key Finding ---%
\begin{frame}{Key Finding: FPGA Offload Candidate}
\begin{alertblock}{Primary Finding}
\texttt{propagate\_to\_next\_surface} kernel: \textbf{63\% of GPU time}, \textbf{93\% warp stalls}
\end{alertblock}

\vspace{0.5em}
\begin{columns}
\column{0.5\textwidth}
\textbf{Why It's Latency-Bound:}
\begin{itemize}
    \item RK4 integration: sequential MAC operations
    \item B-field lookup: dependent memory access
    \item Variable step count: 1--34 steps/track
    \item 62\% wasted GPU cycles due to divergence
\end{itemize}

\column{0.5\textwidth}
\textbf{Why FPGA is Ideal:}
\begin{itemize}
    \item DSP58: native FP32 MAC chains
    \item Pipelined execution (no warp sync)
    \item Dedicated pipelines per track
    \item Variable steps $\neq$ resource waste
\end{itemize}
\end{columns}

\section{Precision Validation}
%==============================================================================%

\begin{frame}{FP32 vs FP64: No Physics Difference}
\begin{columns}
\column{0.5\textwidth}
\textbf{Validation Result:}
\begin{itemize}
    \item Chi$^2$/NDF identical: $1.0042 \pm 0.0020$
    \item Pull distributions: $\sigma$ differs by $<0.003$
    \item All 10,000 tracks tested
\end{itemize}

\vspace{0.5em}
\begin{table}
\scriptsize
\begin{tabular}{lcccc}
    \toprule
    \textbf{Param} & \textbf{FP32 $\sigma$} & \textbf{FP64 $\sigma$} & \textbf{$\Delta\sigma$} \\
    \midrule
    d0 & 0.987 & 0.987 & 0.000 \\
    $\phi$ & 0.980 & 0.980 & 0.000 \\
    q/p & 0.980 & 0.980 & 0.000 \\
    $\theta$ & 0.953 & 0.955 & 0.002 \\
    z0 & 0.987 & 0.987 & 0.000 \\
    \bottomrule
\end{tabular}
\caption{10 GeV muon pull distributions}
\end{table}

\column{0.5\textwidth}
\textbf{Implication for FPGA:}
\begin{itemize}
    \item DSP58 native FP32 $\Rightarrow$ \textbf{No precision loss}
    \item No FP32$\leftrightarrow$FP64 conversion needed
    \item GPU and FPGA produce identical physics
\end{itemize}

\vspace{0.5em}
\begin{block}{Conclusion}
FP32 is physics-equivalent to FP64 for TRACCC Kalman filtering. DSP58 native FP32 is safe for FPGA offloading.
\end{block}
\end{columns}
\end{frame}

\begin{itemize}
    \item 93\% warp stall cycles

    \item SIMT model penalizes divergence

\section{Risk Assessment}
%==============================================================================%

\begin{frame}{Critical Blockers and Risks}
\begin{columns}
\column{0.5\textwidth}
\textbf{CRITICAL: Per-Step Sync Barrier}
\begin{itemize}
    \item CKF requires GPU$\leftrightarrow$FPGA sync per step
    \item 15 surfaces $\times$ sync overhead
    \item If overhead $>$200$\mu$s/step: \textbf{not viable}
    \item \textcolor{acadred}{Must validate before development}
\end{itemize}

\vspace{0.5em}
\textbf{PCIe Latency Analysis:}
\begin{table}
\scriptsize
\begin{tabular}{lcc}
    \toprule
    \textbf{Scenario} & \textbf{Overhead} & \textbf{Status} \\
    \midrule
    Params only & 2.7\% & \cmark OK \\
    Full state & 13\% & Concerning \\
    + Jacobians & 18\% & \xmark Bad \\
    \bottomrule
\end{tabular}
\end{table}

\column{0.5\textwidth}
\textbf{Other Risks:}
\begin{itemize}
    \item DSP58 resource estimates: theoretical
    \item XRT integration complexity
    \item Vitis HLS learning curve
    \item B-field too large for BRAM (139 MB)
\end{itemize}

\vspace{0.5em}
\textbf{Mitigations:}
\begin{itemize}
    \item Cache covariance in HBM2e
    \item Transfer params only (24 bytes/track)
    \item Use V80 HBM for B-field
    \item Start with propagation kernel only
\end{itemize}
\end{columns}
\end{frame}
